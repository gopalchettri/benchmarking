import re
import json

def extract_answer_json(raw_text: str):
    """
    Extract JSON inside <answer>...</answer> and return it as a Python object.
    Raises ValueError if tags or JSON are invalid.
    """
    # 1) Find the <answer>...</answer> block (case-insensitive, multiline)
    match = re.search(r"<answer>(.*?)</answer>", raw_text, re.DOTALL | re.IGNORECASE)
    if not match:
        raise ValueError("No <answer>...</answer> block found in model output")

    answer_block = match.group(1).strip()

    # 2) Parse the JSON inside that block
    try:
        data = json.loads(answer_block)
    except json.JSONDecodeError as e:
        raise ValueError(f"Failed to parse JSON inside <answer> tag: {e}") from e

    return data
